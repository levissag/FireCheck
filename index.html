async function carregarExtintor() { const params = new URLSearchParams(window.location.search); const cod = params.get("cod"); // ex: ext001 if (!cod) { document.getElementById("result").innerText = "Código não informado."; return; } const urlCSV = "https://docs.google.com/spreadsheets/d/e/2PACX-1vTU0F86l8__VCCuZ_eeAu4oDcAgxVR0kiZ2wuckk2O0giZP0JB44lcw1lfZd3gGVdXIZX0OdVzZSCMU/pub?output=csv"; try { const response = await fetch(urlCSV); const csv = await response.text(); // transformar CSV em linhas const linhas = csv.split("\n").map(l => l.trim()); const colunas = linhas[0].split(",").map(c => c.trim()); // cabeçalho // procurar linha do código const alvo = linhas .slice(1) .map(linha => linha.split(",")) .find(c => c[0].toLowerCase() === cod.toLowerCase()); if (!alvo) { document.getElementById("result").innerText = "Extintor não encontrado."; return; } // montar objeto simples const ext = {}; colunas.forEach((nome, i) => { ext[nome] = alvo[i] || ""; }); // exibir document.getElementById("result").innerHTML = `

Extintor ${ext.Codextin}

Esquadrão: ${ext.Esquadrão}

Prédio: ${ext.Prédio}

Local: ${ext.Local}

Tipo: ${ext.Tipo}

Capacidade: ${ext.Capacidade}

Vistoriador: ${ext.Vistoriador}

Data da Vistoria: ${ext.DataVist}

Obs: ${ext.Obs}

`; } catch (e) { document.getElementById("result").innerText = "Erro ao carregar os dados."; console.error(e); } } carregarExtintor();
