async function loadExtintor() { const url = "https://docs.google.com/spreadsheets/d/1iMx5Mc4-ayuoE95p5Kg4yqTp8peNHWNYRdeszSdPK8I/gviz/tq?tqx=out:json"; const response = await fetch(url); const text = await response.text(); // Extrai o JSON corretamente — sem substring fixa! const jsonText = text.substring(text.indexOf("{"), text.lastIndexOf("}") + 1); const data = JSON.parse(jsonText); const rows = data.table.rows.map(r => r.c.map(c => c ? c.v : "")); const header = data.table.cols.map(c => c.label); const items = rows.map(r => { let obj = {}; header.forEach((h, i) => obj[h] = r[i]); return obj; }); const urlParams = new URLSearchParams(window.location.search); const cod = urlParams.get("cod"); if (!cod) { document.getElementById("conteudo").innerHTML = "Código não informado"; return; } const ext = items.find(x => x.Codextin == cod); if (!ext) { document.getElementById("conteudo").innerHTML = "Extintor não encontrado"; return; } document.getElementById("conteudo").innerHTML = `

Dados do Extintor

Código: ${ext.Codextin}

Esquadrão: ${ext.Esquadrão}

Prédio: ${ext.Prédio}

Local: ${ext.Local}

Tipo: ${ext.Tipo}

Capacidade: ${ext.Capacidade}

Vistoriador: ${ext.Vistoriador}

Data da Vistoria: ${ext.DataVist}

Observações: ${ext.Obs}

`; } loadExtintor();
