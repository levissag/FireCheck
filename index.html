async function carregarDadosExtintor() { const urlParams = new URLSearchParams(window.location.search); const cod = urlParams.get("cod"); if (!cod) { document.getElementById("conteudo").innerHTML = "

Nenhum código informado.

"; return; } const csvUrl = "https://docs.google.com/spreadsheets/d/e/2PACX-1vTU0F86l8__VCCuZ_eeAu4oDcAgxVR0kiZ2wuckk2O0giZP0JB44lcw1lfZd3gGVdXIZX0OdVzZSCMU/pub?output=csv"; try { const response = await fetch(csvUrl); const csvText = await response.text(); const lines = csvText.split("\n").filter(l => l.trim().length > 0); const header = lines.shift().split(",").map(h => h.trim()); const data = lines.map(line => { const values = line.split(",").map(v => v.trim()); const obj = {}; header.forEach((h, i) => { obj[h] = values[i] || ""; }); return obj; }); const ext = data.find(item => item.Codextin.toUpperCase() === cod.toUpperCase()); if (!ext) { document.getElementById("conteudo").innerHTML = `

Extintor ${cod} não encontrado.

`; return; } document.getElementById("conteudo").innerHTML = `

Dados do Extintor

Código: ${ext.Codextin}

Esquadrão: ${ext.Esquadrão}

Prédio: ${ext.Prédio}

Local: ${ext.Local}

Tipo: ${ext.Tipo}

Capacidade: ${ext.Capacidade}

Vistoriador: ${ext.Vistoriador}

Data da Vistoria: ${ext.DataVist}

Observações: ${ext.Obs}

`; } catch (e) { console.error("Erro ao carregar:", e); document.getElementById("conteudo").innerHTML = "

Erro ao carregar dados.

"; } } carregarDadosExtintor();
