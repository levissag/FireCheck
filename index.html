//--------------------------------------------------- // 1) PEGAR O CÓDIGO DA URL (?cod=EXT001) //--------------------------------------------------- const urlParams = new URLSearchParams(window.location.search); const cod = urlParams.get("cod"); if (!cod) { document.getElementById("result").innerHTML = "Código não informado na URL."; throw new Error("Código não informado."); } //--------------------------------------------------- // 2) URL DO CSV PUBLICADO DO GOOGLE SHEETS //--------------------------------------------------- const CSV_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vTU0F86l8__VCCuZ_eeAu4oDcAgxVR0kiZ2wuckk2O0giZP0JB44lcw1lfZd3gGVdXIZX0OdVzZSCMU/pub?output=csv"; //--------------------------------------------------- // 3) FUNÇÃO PARA PROCESSAR O CSV //--------------------------------------------------- function parseCSV(csvText) { // Remove BOM e linhas vazias csvText = csvText.replace(/^\uFEFF/, "").trim(); const lines = csvText.split("\n").map(l => l.trim()).filter(l => l.length > 0); const headers = lines[0].split(",").map(h => h.trim()); const data = []; for (let i = 1; i < lines.length; i++) { const obj = {}; const row = lines[i].split(","); headers.forEach((h, index) => { obj[h] = row[index] ? row[index].trim() : ""; }); data.push(obj); } return data; } //--------------------------------------------------- // 4) BUSCAR O CSV E EXIBIR O EXTINTOR //--------------------------------------------------- fetch(CSV_URL) .then(response => response.text()) .then(csv => { const lista = parseCSV(csv); // Procurar o extintor ignorando maiúsc/minúsc const ext = lista.find(e => e.Codextin && e.Codextin.toLowerCase() === cod.toLowerCase() ); if (!ext) { document.getElementById("result").innerHTML = "Extintor não encontrado."; return; } // Montar o HTML final const html = `

Dados do Extintor

Código: ${ext.Codextin}

Esquadrão: ${ext.Esquadrao || ext.Esquadrão || ""}

Prédio: ${ext.Predio || ext.Prédio || ""}

Local: ${ext.Local}

Tipo: ${ext.Tipo}

Capacidade: ${ext.Capacidade}

Vistoriador: ${ext.Vistoriador}

Data da Vistoria: ${ext.DataVist}

Observações: ${ext.Obs}

`; document.getElementById("result").innerHTML = html; }) .catch(err => { document.getElementById("result").innerHTML = "Erro ao carregar dados."; console.error(err); });
